<script>
/* FALLING FLOWERS AND HEARTS */

const symbols = [
  '♡', '♥', '💚', '🌿', '🌸',
  '✿', '❀', '୨୧', '✧', '🍃'
];

function createParticle() {
  const particle = document.createElement('span');

  particle.className = 'particle';

  particle.textContent =
    symbols[Math.floor(Math.random() * symbols.length)];

  particle.style.left = Math.random() * 100 + 'vw';

  particle.style.fontSize =
    (12 + Math.random() * 15) + 'px';

  particle.style.animationDuration =
    (6 + Math.random() * 7) + 's';

  document.getElementById('particles')
    .appendChild(particle);

  setTimeout(() => particle.remove(), 14000);
}

setInterval(createParticle, 350);


/* OPEN AND CLOSE LOVE LETTER */

function openLetter() {
  const letter = document.getElementById('letter');

  letter.classList.toggle('show');

  const button = document.querySelector('.card .main-btn');

  button.textContent = letter.classList.contains('show')
    ? 'CLOSE YOUR LETTER ♡'
    : 'OPEN YOUR LETTER 💌';

  if (letter.classList.contains('show')) {
    letter.scrollIntoView({
      behavior: 'smooth',
      block: 'nearest'
    });

    burst();
  }
}


/* SECRET SURPRISE */

function openSecret() {
  const secret = document.getElementById('secret');

  secret.classList.toggle('show');

  const button = document.querySelector('.surprise .main-btn');

  button.textContent = secret.classList.contains('show')
    ? 'CLOSE MY SECRET ♡'
    : 'UNLOCK MY SECRET ♡';

  if (secret.classList.contains('show')) {
    secret.scrollIntoView({
      behavior: 'smooth',
      block: 'nearest'
    });

    burst();
  }
}


/* YOUTUBE MUSIC STARTING AT 1:04 */

let musicPlaying = false;

function toggleMusic() {
  const player = document.getElementById('youtube-player');
  const button = document.getElementById('musicButton');

  if (!musicPlaying) {

    player.innerHTML = `
      <iframe
        src="https://www.youtube.com/embed/dg3VDvHtX8Q?start=64&autoplay=1&rel=0"
        title="Our Little Song"
        allow="autoplay; encrypted-media; picture-in-picture"
        referrerpolicy="strict-origin-when-cross-origin"
        allowfullscreen>
      </iframe>
    `;

    button.textContent = 'Ⅱ STOP';
    musicPlaying = true;

  } else {

    player.innerHTML = '';

    button.textContent = '▶ PLAY';
    musicPlaying = false;

  }
}


/* LITTLE FLOWER AND HEART BURST */

function burst() {
  for (let i = 0; i < 22; i++) {
    setTimeout(createParticle, i * 45);
  }
}
